### Users Table

sql
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    avatar_url VARCHAR(255),
    bio TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    last_login TIMESTAMP WITH TIME ZONE,
    is_active BOOLEAN DEFAULT TRUE,
    is_admin BOOLEAN DEFAULT FALSE,
    two_factor_enabled BOOLEAN DEFAULT FALSE,
    two_factor_secret VARCHAR(255),
    wallet_address VARCHAR(255),
    reset_token VARCHAR(255),
    reset_token_expires TIMESTAMP WITH TIME ZONE
);


### UserStats Table

sql
CREATE TABLE user_stats (

    defense_success_rate INTEGER DEFAULT 0,
    total_upgrades INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);


### Missions Table

sql
CREATE TABLE missions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(100) NOT NULL,
    description TEXT NOT NULL,
    difficulty VARCHAR(20) NOT NULL CHECK (difficulty IN ('Easy', 'Medium', 'Hard', 'Extreme')),
    type VARCHAR(20) NOT NULL CHECK (type IN ('Legal', 'Illegal')),
    base_reward INTEGER NOT NULL,
    time_required INTEGER NOT NULL, -- in seconds
    min_level INTEGER DEFAULT 1,
    success_rate INTEGER DEFAULT 70, -- base success rate percentage
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);


### UserMissions Table

sql
CREATE TABLE user_missions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    mission_id UUID NOT NULL REFERENCES missions(id) ON DELETE CASCADE,
    status VARCHAR(20) NOT NULL CHECK (status IN ('In Progress', 'Completed', 'Failed', 'Aborted')),
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE,
    reward_earned INTEGER,
    experience_gained INTEGER,
    UNIQUE(user_id, mission_id, started_at)
);

### Vulnerabilities Table

sql

### PvPAttacks Table

sql
CREATE TABLE pvp_attacks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    attacker_id UUID NOT NULL REFERENCES users(id),
    defender_id UUID NOT NULL REFERENCES users(id),
    result VARCHAR(20) NOT NULL CHECK (result IN ('Success', 'Failed')),
    crypto_stolen INTEGER DEFAULT 0,
    reputation_change INTEGER DEFAULT 0,
    attack_timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    CHECK (attacker_id != defender_id)
);

## Database Relationships Explained

1. **Users to UserStats**: One-to-one relationship - each user has one set of stats
2. **Users to Upgrades**: One-to-many relationship - users can own multiple upgrades
3. **Users to SecuritySystems**: One-to-many relationship - users can have multiple security systems
4. **Users to Missions**: One-to-many relationship - users can complete multiple missions
5. **Users to PvPAttacks**: One-to-many relationship - users can initiate or defend in multiple PvP attacks
6. **Users to Transactions**: One-to-many relationship - users can have multiple transactions
7. **Users to Notifications**: One-to-many relationship - users can receive multiple notifications
8. **Users to Achievements**: One-to-many relationship - users can earn multiple achievements
9. **Missions to MissionRewards**: One-to-many relationship - missions can give multiple types of rewards
10. **Upgrades to UpgradeEffects**: One-to-many relationship - upgrades can have multiple effects at different levels
11. **SecuritySystems to Vulnerabilities**: One-to-many relationship - security systems can have multiple vulnerabilities
